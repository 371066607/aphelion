```js
/* ---------- 建造队列(纯函数): 并行上限3, 按序完工 ----------
   q 项: {bid,x,y,remain}。返回 {queue, done:[{bid,x,y}]} */
function queueTick(queue, dt){
  var done = [];
  var active = 0;
  var out = queue.map(function(item){
    if (active >= 3) return item;          // 超过并行数的原地等待
    active++;
    var r = item.remain - dt;
    if (r <= 0){ done.push({bid:item.bid, x:item.x, y:item.y}); return null; }
    return Object.assign({}, item, {remain:r});
  }).filter(Boolean);
  return { queue: out, done: done };
}
```

main.js 侧：
- state 加 `s.colony.buildQueue = []`（loadColony 兼容旧档默认 []）
- `tryPlace()` 改为：扣费 → `s.colony.buildQueue.push({bid,x,y,remain:def.buildTime})` → saveColony → 不再直接 placeBuildingEntity
- updateHome 每帧：`var r = APH.Colony.queueTick(s.colony.buildQueue, dt); s.colony.buildQueue = r.queue; r.done.forEach(d => { placeBuildingEntity(d.bid,d.x,d.y); U.emit('built',{id:d.bid}); })`

function canUpgrade(b, def, research){
  if (!def.upg) return {ok:false, why:'不可升级'};
  var lv = b.lv||1;
  if (lv >= (def.upg.maxLv+1)) return {ok:false, why:'已达最高等级'};
  var cost = upgradeCost(def, lv);
  if (research < cost) return {ok:false, why:'研究点不足(需'+cost+')'};

}

/* T13 炮塔: 每座塔独立冷却, 射程240, 伤害=8+8×(lv-1) */
function turretStep(turret, enemies, dt){
  turret.cd = (turret.cd||0) - dt;
  if (turret.cd > 0) return;
  var dmg = 8 + CFG_TURRET_DMG_PER_LV * ((turret.lv||1)-1);   // 常量进 config.js
  for (var i=0;i<enemies.length;i++){
    var en = enemies[i];
    if (en.dead) continue;
    if (U.dst(turret.x,turret.y,en.x,en.y) < 240){
      en.hp -= dmg; en.hitFlash = 0.1;
      turret.cd = 1.2;
      U.emit('enemyHit', en);
      if (en.hp <= 0) killEnemy(en);
      return;
    }
  }
}
function soldierCount(barracks){ 
  return barracks.reduce(function(a,b){ return a + 2*((b.lv||1)); }, 0);

**形状与质感**：圆角 18px(基础)/50px(胶囊按钮)；2px 实线边框；主按钮 3D 堆叠阴影 `0 5px 0 0 #bdaea0`（hover 6px / active 1px + 下压 translateY(2px)）；卡片柔和阴影 `0 3px 10px rgba(61,52,40,.1)`。

**字体**：`<link>` 引入 Google Fonts `Nunito:wght@400..900 + Noto Sans SC:wght@400;500;700`；正文 500 / 按钮 700 / 数字强调 900。禁用 light 细体和等宽体。

**动效**：全部过渡 `0.25s cubic-bezier(0.4,0,0.2,1)`；hover 上浮 translateY(-1~-2px)；active 下压 2px。
